% \cventry[logo]{date}{body}: date at the top of the left column, the optional
% logo centred in the space below it, a thin vertical rule, then the body. A
% short entry grows to fit the date and the logo.
% The gap below is a bare \vskip, which a following heading's own space
% absorbs; \vspace would hide it and the two would stack.
\newsavebox{\cvbox}
\newsavebox{\cvlogobox}
\newcommand{\cventry}[3][]{%
  \sbox{\cvbox}{\begin{minipage}[t]{\bodyw}\raggedright\small #3\end{minipage}}%
  \ifx\relax#1\relax\sbox{\cvlogobox}{}\else
    \sbox{\cvlogobox}{\includegraphics[height=0.8cm,width=2.1cm,keepaspectratio]{#1}}\fi
  \noindent
  \makebox[\datew][c]{\raisebox{\dimexpr(\ht\cvbox-\dp\cvbox)/2-\ht\cvlogobox/2\relax}[0pt][0pt]{\usebox{\cvlogobox}}}\hspace{0.15cm}%
  {\color{rulegray}\vrule width 0.4pt height \ht\cvbox depth \dp\cvbox}\hspace{0.25cm}%
  \usebox{\cvbox}\par\vskip 6pt\relax}
