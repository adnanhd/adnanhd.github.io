\newcommand{\cventry}[3][]{%
  \sbox{\cvbox}{\begin{minipage}[t]{\bodyw}\raggedright #3\end{minipage}}%
  \noindent
  \begin{minipage}[t]{\datew}\raggedright #2%
    \ifx\relax#1\relax\else\\[5pt]\includegraphics[height=0.72cm,width=2.1cm,keepaspectratio]{#1}\fi

  {\color{rulegray}\vrule width 0.4pt height \ht\cvbox depth \dp\cvbox}\hspace{0.25cm}%
  \usebox{\cvbox}\par\vspace{9pt}}
